Data sources

The chunks in this section download each source as published into data/raw/. By default, they do not run when the notebook is rendered. Download them by hand by setting the eval chunk option to true. Each download overwrites the files in data/raw/ and rewrites data/raw/manifest.csv.

Code 
raw_dir <- "data/raw"

# One manifest row per downloaded file. The Council reissues its workbooks
# under the same file name, so the download time and checksum are what tell
# two copies of a fiscal year apart.
manifest_row <- function(source, item, path, url, source_modified = NA, rows = NA) {
  tibble(
    source          = source,
    item            = as.character(item),
    path            = path,
    url             = url,
    fetched_at      = format(Sys.time(), "%Y-%m-%dT%H:%M:%SZ", tz = "UTC"),
    source_modified = as.character(source_modified),
    bytes           = file.size(path),
    rows            = as.integer(rows),
    sha256          = unname(tools::sha256sum(path))
  )
}

# Saves a file byte-for-byte and returns the server's Last-Modified date.
download_raw <- function(url, path) {
  dir.create(dirname(path), recursive = TRUE, showWarnings = FALSE)
  modified <- request(url) |>
    req_retry(max_tries = 3) |>
    req_perform(path = path) |>
    resp_header("last-modified")
  if (is.null(modified)) return(NA)
  format(curl::parse_date(modified), "%Y-%m-%dT%H:%M:%SZ", tz = "UTC")
}

manifest_open_data <- imap(open_data, \(id, name) {
  url  <- paste0("https://data.cityofnewyork.us/d/", id)
  path <- file.path(raw_dir, "open_data", paste0(name, "_", id, ".csv"))
  dir.create(dirname(path), recursive = TRUE, showWarnings = FALSE)
  # alias = "drop" keeps the API field names (mocs_id, council_member) instead
  # of the display labels.
  data <- soc_read(url, alias = "drop")
  updated <- format(attr(data, "data_last_updated"), "%Y-%m-%dT%H:%M:%SZ", tz = "UTC")
  # A Socrata url column arrives as a nested url/description pair. The portal's
  # own CSV export writes the url alone, so that is what is written here.
  data <- mutate(data, across(where(is.data.frame), \(x) x$url))
  # The portal publishes dates as midnight with no time zone. Written as the
  # date alone, so a UTC conversion cannot move them.
  data <- mutate(data, across(where(is.POSIXct), as_date))
  write_csv(data, path, na = "")
  manifest_row("open_data", id, path, url, updated, nrow(data))
}) |>
  list_rbind()

manifest_workbooks <- map(2009:2026, \(fy) {
  file <- paste0("funded_disclosure_FY", fy, if (fy <= 2013) ".xls" else ".xlsx")
  url  <- paste0("https://rnd.council.nyc.gov/src/assets/excelFund/", file)
  path <- file.path(raw_dir, "workbooks", file)
  modified <- download_raw(url, path)
  manifest_row("workbook", fy, path, url, modified)
}) |>
  list_rbind()

Schedule C

The Adopted Expense Budget Adjustment Summary, FY2008 through FY2027. There is no index of these PDFs: each is linked from its fiscal year’s page at council.nyc.gov/budget/fy####/. The FY2011 page links two documents, a Schedule C and the Changes Pursuant to Section 254; both are downloaded.

Code 
manifest_schedule_c <- map(2008:2027, \(fy) {
  page  <- paste0("https://council.nyc.gov/budget/fy", fy, "/")
  links <- read_html(page) |> html_elements("a")
  links <- tibble(
    text = html_text2(links),
    url  = xml2::url_absolute(html_attr(links, "href"), page)
  ) |>
    filter(str_detect(url, regex("\\.pdf$", ignore_case = TRUE)))

  schedule_c <- links |>
    filter(str_detect(text, regex("schedule c", ignore_case = TRUE))) |>
    slice(1) |>
    mutate(file = paste0("Schedule_C_FY", fy, ".pdf"))
  # From FY2020 on, "Section 254" on these pages titles capital budget
  # documents, so it is only followed for FY2011.
  section_254 <- links |>
    filter(fy == 2011, str_detect(text, regex("section 254", ignore_case = TRUE))) |>
    slice(1) |>
    mutate(file = paste0("Schedule_C_FY", fy, "_Section_254.pdf"))

  if (nrow(schedule_c) == 0) stop("No Schedule C link on ", page, call. = FALSE)

  bind_rows(schedule_c, section_254) |>
    pmap(\(text, url, file) {
      path <- file.path(raw_dir, "schedule_c", file)
      modified <- download_raw(url, path)
      manifest_row("schedule_c", fy, path, url, modified)
    }) |>
    list_rbind()
}) |>
  list_rbind()

Seven workbooks, FY2009 through FY2014 and FY2016, have no fiscal year column. For those the year is taken from the file name.

The FY2024 workbook has seventeen rows below its last award: twelve empty rows and five rows of totals. Rows with neither a source nor a legal name are dropped on read. No other workbook has rows meeting this condition.

Code 
is_blank <- function(x) is.na(x) | str_trim(x) == ""

read_workbook <- function(path, fy) {
  # trim_ws = FALSE keeps leading and trailing spaces in the values as published.
  data <- read_excel(path, sheet = 1, col_types = "text", trim_ws = FALSE)

  unmapped <- setdiff(names(data), workbook_crosswalk$workbook_header)
  if (length(unmapped) > 0) {
    stop(basename(path), " has a header the crosswalk does not cover: ",
         paste(unmapped, collapse = ", "), call. = FALSE)
  }
  names(data) <- workbook_crosswalk$open_data_name[match(names(data), workbook_crosswalk$workbook_header)]

  if (!"fiscal_year" %in% names(data)) data$fiscal_year <- as.character(fy)

  filter(data, !(is_blank(source) & is_blank(legal_name_of_organization)))
}

workbooks <- map2(workbook_paths, workbook_fy, read_workbook) |>
  list_rbind() |>
  select(all_of(unique(workbook_crosswalk$open_data_name)))

Comparing the workbooks with Open Data

The two sources cover the same designations for FY2009 through FY2021. A workbook row and an Open Data row are counted as the same award when fiscal year, source, organization name, and amount all agree. Before comparing, names are lowercased, punctuation is removed, and repeated spaces are collapsed; sources are lowercased and their spaces collapsed; amounts are written to two decimal places. Where a key appears more than once in a year, rows are paired off one for one.

Code 
match_keys <- function(data) {
  data |>
    transmute(
      fiscal_year = as.integer(fiscal_year),
      amount      = as.numeric(amount),
      key = paste(
        str_squish(str_to_lower(source)),
        str_squish(str_remove_all(str_to_lower(legal_name_of_organization), "[[:punct:]]")),
        sprintf("%.2f", amount),
        sep = "|"
      )
    )
}

workbook_keys  <- match_keys(workbooks)
open_data_keys <- match_keys(open_data_funding)

source_totals <- full_join(
  summarise(workbook_keys, workbook_rows = n(), workbook_amount = sum(amount), .by = fiscal_year),
  summarise(open_data_keys, open_data_rows = n(), open_data_amount = sum(amount), .by = fiscal_year),
  by = "fiscal_year"
)

source_matches <- full_join(
  count(workbook_keys, fiscal_year, key, name = "in_workbook"),
  count(open_data_keys, fiscal_year, key, name = "in_open_data"),
  by = c("fiscal_year", "key")
) |>
  replace_na(list(in_workbook = 0, in_open_data = 0)) |>
  summarise(
    matched        = sum(pmin(in_workbook, in_open_data)),
    workbook_only  = sum(pmax(in_workbook - in_open_data, 0)),
    open_data_only = sum(pmax(in_open_data - in_workbook, 0)),
    .by = fiscal_year
  )

source_comparison <- source_totals |>
  left_join(source_matches, by = "fiscal_year") |>
  arrange(fiscal_year) |>
  # Open Data stops at FY2021, so there is nothing to match against after it.
  mutate(across(c(matched, workbook_only, open_data_only), \(x) if_else(is.na(open_data_rows), NA, x)))

source_comparison |>
  transmute(
    `Fiscal year`    = paste0("FY", fiscal_year),
    `Workbook rows`  = workbook_rows,
    `Open Data rows` = open_data_rows,
    Matched          = matched,
    `Workbook only`  = workbook_only,
    `Open Data only` = open_data_only,
    `Workbook $ minus Open Data $` = round(workbook_amount - open_data_amount)
  ) |>
  mutate(across(-`Fiscal year`, \(x) if_else(is.na(x), "", format(x, big.mark = ",", trim = TRUE)))) |>
  cil_table(full_height = TRUE)
